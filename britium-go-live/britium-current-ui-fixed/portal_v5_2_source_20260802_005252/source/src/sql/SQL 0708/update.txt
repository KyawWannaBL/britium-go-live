-- ---------- Role helpers ----------
create or replace function public.be_current_role()
returns text
language sql
stable
security definer
set search_path = public
as $$
  select coalesce(
    (select lower(replace(p.role, '-', '_')) from public.profiles p where p.id = auth.uid()),
    lower(replace(auth.jwt() ->> 'role', '-', '_')),
    lower(replace(auth.jwt() ->> 'app_role', '-', '_')),
    ''
  );
$$;

create or replace function public.be_has_role(allowed text[])
returns boolean
language sql
stable
security definer
set search_path = public
as $$
  select public.be_current_role() = any(allowed)
     or coalesce(auth.jwt() ->> 'email','') in ('md@britiumexpress.com','sai@britiumexpress.com');
$$;
